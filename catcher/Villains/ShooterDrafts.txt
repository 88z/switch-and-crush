//решаем уравнение пересечения траектории цели и окружности с центром в месте выстрела и скорости пули
//корни -- точки попадания
// если цель стоит на месте, то точка попадания -- координаты цели

// для тестов ресуем линию между точкой выстрела и точкой попадания а вектор скорости задаем -- вперед

func shoot(timeOfBulletFly: CGFloat, scatter:CGFloat) {
    let debug = false
    guard let camera = scene.camera, let targetBody = target.physicsBody else {
        return
    }
    let visibleRect = CGRect(x: camera.position.x - scene.size.width/2,
                             y: camera.position.y - scene.size.height/2, width: scene.size.width, height: scene.size.height)
    
    //точка выстрела
    let shootingPoint = randomPointOn(visibleRect)
    if (debug) {
        let shootTestSpot = SKShapeNode(circleOfRadius: 10)
        shootTestSpot.fillColor = .green
        shootTestSpot.position = shootingPoint
        target.scene?.addChild(shootTestSpot)
    }
    
    //модуль скорости выстрела
    let shootingSpeed = CGFloat(1000)
    
    
    let targetVelocity = targetBody.velocity
//        let targetVelocity = CGVector(dx:0, dy: -100)
    
    //собираем все нужное для квадратного уравнения и решаем
    
    let a = targetVelocity.dx/targetVelocity.dy
    let b = a*target.position.y - target.position.x
    
    let a1 = Double(a*a+1)
    let b1 = Double(-2*a*b - 2*a*shootingPoint.x - 2*shootingPoint.y)
    let c1 = Double(b*b + 2*b*shootingPoint.x + shootingPoint.x*shootingPoint.x + shootingPoint.y*shootingPoint.y - shootingSpeed*shootingSpeed)
    
    let y12: [Double] = quadraticRealSolve(a: a1, b: b1, c: c1)
    
    // убираем нерпавильные корни.
    // в этом решении я сомневаюсь, потому что непонятно, как определять правильный корень или нет
    // текущее сработает при одинаковых моулях скоростей пули и цели
    // если пули будут лететь не туда, надо будет проверять эту штуку
    
    guard let y = y12.filter ({ (yi) -> Bool in
        
        // если скорость отрицательная, то правильный корень должен быть меньше меньше разности позиции и скорости
        //если скорость положительная, то правильный корень должен быть больше суммы скорости и позиции
        if targetVelocity.dy > 0 {
            return yi >= Double(target.position.y + targetVelocity.dy)
        } else {
            return yi <= Double(target.position.y + targetVelocity.dy)
        }
        
    }).first else {
        // скорость пули маленькая, чтобы попасть
        print("miss")
        return
        
    }
    
    let hitPoint = CGPoint(x: a*CGFloat(y)-b, y: CGFloat(y))
    
    if (debug) {
        let testNode = SKShapeNode()
        let testPath = CGMutablePath()
        testPath.move(to: shootingPoint)
        testPath.addLine(to: hitPoint)
        testNode.path = testPath
        testNode.strokeColor = .red
        target.scene?.addChild(testNode)
    } else {
        //направление вектора выстрела
        let shootingVector = CGVector(dx: hitPoint.x - shootingPoint.x, dy: hitPoint.y - shootingPoint.y)
        
        let bullet = Bullet(collisionBitMask: bulletCollisionBitMask, categoryBitMask: bulletCategoryBitMask, contactBitMask:bulletContactBitMask)
        
        bullet.position = shootingPoint
        bullet.physicsBody?.velocity = CGVector(shootingVector, changeLenTo: shootingSpeed)
        bullets.append(bullet)
        scene.addChild(bullet)
    }
    
}

func shoot1(timeOfBulletFly: CGFloat, scatter:CGFloat) {
    guard let camera = scene.camera, let targetBody = target.physicsBody else {
        return
    }
    let visibleRect = CGRect(x: camera.position.x - scene.size.width/2,
                             y: camera.position.y - scene.size.height/2, width: scene.size.width, height: scene.size.height)
    print (targetBody.velocity)
    //точка выстрела
    let shootingPoint = randomPointOn(visibleRect)
    
    
    //через какое расстояние от текущего положения точки будет попадание
    let hitPointDelta = CGVector(targetBody.velocity, changeLenTo: CGVectorLen(targetBody.velocity)*timeOfBulletFly)
    
    //точка попадания
    let hitPoint = randomPointInsideSquareWith(center: CGPoint(x: target.position.x+hitPointDelta.dx, y: target.position.y + hitPointDelta.dy), side: scatter)
    
    //направление вектора выстрела
    let shootingVector = CGVector(dx: hitPoint.x - shootingPoint.x, dy: hitPoint.y - shootingPoint.y)
    
    //длина вектора выстрела
    let shootingVectorLen = CGPointDistance(from: hitPoint, to: shootingPoint)/timeOfBulletFly
    
    let bullet = Bullet(collisionBitMask: bulletCollisionBitMask, categoryBitMask: bulletCategoryBitMask, contactBitMask:bulletContactBitMask)
    
    bullet.position = shootingPoint
    bullet.physicsBody?.velocity = CGVector(shootingVector, changeLenTo: shootingVectorLen)
    bullets.append(bullet)
    scene.addChild(bullet)
}
